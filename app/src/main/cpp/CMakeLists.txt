cmake_minimum_required(VERSION 3.22.1)
project(lanu_ai LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
# Even test APKs need optimized inference; keep symbols for diagnostics.
set(CMAKE_C_FLAGS_DEBUG "-O3 -g" CACHE STRING "" FORCE)
set(CMAKE_CXX_FLAGS_DEBUG "-O3 -g" CACHE STRING "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
include(FetchContent)
FetchContent_Declare(llama URL https://github.com/ggml-org/llama.cpp/archive/d8d92973c8c21bb4dd5d592f0fec26fe2cd1e654.tar.gz URL_HASH SHA256=c376d74f3498fab2b9e266b4aa75ce8eec1f13e4e9228492e015edcfa0d05962)
FetchContent_MakeAvailable(llama)
if(ANDROID)
    add_library(lanu-ai SHARED native.cpp generation.cpp)
    target_link_libraries(lanu-ai PRIVATE llama android log)
    target_link_options(lanu-ai PRIVATE "-Wl,-z,max-page-size=16384")
else()
    # Quality validation uses the exact JNI generator, not server defaults.
    add_executable(lanu-ai-probe host_probe.cpp generation.cpp)
    target_link_libraries(lanu-ai-probe PRIVATE llama)
endif()
